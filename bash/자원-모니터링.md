# 시스템 자원 모니터링 명령어 모음

## GPU 모니터링

```bash
# ── nvidia-smi (기본 내장) ──
nvidia-smi                             # 스냅샷: GPU 사용률, VRAM, 온도, 프로세스
watch -n 1 nvidia-smi                  # 1초마다 갱신 (실시간 모니터링)
nvidia-smi -l 2                        # 2초 간격 자동 갱신 (built-in loop)

# 특정 정보만 뽑기
nvidia-smi --query-gpu=name,temperature.gpu,utilization.gpu,memory.used,memory.total --format=csv
nvidia-smi --query-compute-apps=pid,name,used_memory --format=csv   # GPU 점유 프로세스

# ── nvtop (htop 스타일 GPU 모니터) ──
sudo apt install nvtop
nvtop                                  # 인터랙티브 UI + 사용률 그래프

# ── gpustat (간결한 GPU 상태) ──
pip install gpustat
gpustat                                # 한 줄 요약
gpustat -i 1                           # 1초 간격 갱신
gpustat -cp                            # 프로세스 이름 + PID 표시
```

### GPU 도구 비교

| 도구 | 설치 | 특징 |
|------|------|------|
| `nvidia-smi` | 기본 내장 | 가장 상세, 스냅샷 형태 |
| `nvtop` | apt | htop 스타일 인터랙티브 UI, 그래프 |
| `gpustat` | pip | 간결한 한 줄 요약, 스크립트에 유용 |

## CPU 모니터링

```bash
# ── top (기본 내장) ──
top                                    # 실시간 프로세스 목록 + CPU/메모리
# top 안에서: 1 → 코어별 CPU 보기, q → 종료, k → 프로세스 kill

# ── htop (향상된 top) ──
sudo apt install htop
htop                                   # 컬러 UI + 마우스 지원 + 트리뷰

# ── CPU 정보 확인 ──
lscpu                                  # CPU 모델, 코어 수, 아키텍처
nproc                                  # 사용 가능한 CPU 코어 수
cat /proc/cpuinfo | head -30           # 상세 CPU 정보

# ── CPU 사용률만 빠르게 ──
mpstat                                 # 전체 CPU 사용률 (sysstat 패키지)
mpstat -P ALL 1                        # 코어별 사용률, 1초 간격
```

## 메모리 (RAM) 모니터링

```bash
# ── free (기본 내장) ──
free -h                                # 전체/사용중/여유 메모리 (읽기 쉬운 단위)
free -h -s 2                           # 2초 간격 갱신
watch -n 1 free -h                     # 1초마다 갱신

# ── 상세 메모리 정보 ──
cat /proc/meminfo                      # 커널이 보고하는 상세 메모리
vmstat 1                               # 1초 간격 가상메모리 통계

# ── 프로세스별 메모리 사용량 ──
ps aux --sort=-%mem | head -10         # 메모리 많이 쓰는 프로세스 TOP 10
```

## 디스크 모니터링

```bash
# ── 전체 디스크 용량 ──
df -h                                  # 마운트된 디스크별 사용량
df -h /home                            # 특정 경로의 디스크 정보

# ── 폴더별 용량 확인 ──
du -sh *                               # 현재 폴더 내 항목별 크기
du -sh ~/project/*                     # 특정 폴더 하위 항목
du -sh ~ --exclude='.cache'            # 특정 폴더 제외

# 큰 폴더/파일 찾기
du -h --max-depth=1 ~ | sort -rh | head -10    # 홈 아래 큰 폴더 TOP 10
find / -type f -size +1G 2>/dev/null           # 1GB 이상 파일 찾기

# ── 디스크 I/O ──
sudo apt install iotop
sudo iotop                             # 프로세스별 디스크 읽기/쓰기 실시간
iostat -x 1                            # 디스크별 I/O 통계, 1초 간격
```

## 네트워크 모니터링

```bash
# ── 기본 네트워크 정보 ──
ip addr                                # IP 주소 확인
ip route                               # 라우팅 테이블

# ── 포트 & 연결 확인 ──
ss -tuln                               # 열린 포트 목록 (TCP/UDP)
ss -tp                                 # 연결된 프로세스 확인

# ── 실시간 트래픽 ──
sudo apt install nethogs
sudo nethogs                           # 프로세스별 네트워크 사용량

sudo apt install iftop
sudo iftop                             # 인터페이스별 트래픽 실시간
```

## 종합 모니터링 도구

```bash
# ── glances (올인원 모니터링) ──
pip install glances
glances                                # CPU + RAM + 디스크 + 네트워크 + GPU 한 화면에

# ── btop (최신 htop 대체) ──
sudo apt install btop
btop                                   # 화려한 UI, CPU/RAM/디스크/네트워크 통합
```

### 종합 도구 비교

| 도구 | 설치 | 범위 | 특징 |
|------|------|------|------|
| `htop` | apt | CPU + RAM + 프로세스 | 가볍고 빠름, 가장 보편적 |
| `btop` | apt | CPU + RAM + 디스크 + 네트워크 | 화려한 UI, htop 상위호환 |
| `glances` | pip | 전부 (GPU 포함) | Python 기반, 웹 UI 가능 |
| `nvtop` | apt | GPU 전용 | GPU 모니터링 최고 |

## 딥러닝 학습 시 추천 조합

```bash
# 터미널 1: GPU 모니터링
nvtop

# 터미널 2: CPU + RAM
htop

# 터미널 3: 디스크 용량 (체크포인트 쌓이는지 확인)
watch -n 10 'du -sh ~/checkpoints/* | tail -5 && echo "---" && df -h /home'

# 한 줄로 GPU + RAM 동시 확인
watch -n 1 'gpustat && echo "---" && free -h'
```

> 💡 **팁**: 학습 돌릴 때 VRAM OOM 뜨면 `nvidia-smi`로 다른 프로세스가 GPU 잡고 있는지 먼저 확인!
